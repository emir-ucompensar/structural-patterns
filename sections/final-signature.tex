% final-signature.tex - Add a non-invasive mark to the last page footer.
%
% The mark lives in the footer layer rather than the document body. This avoids
% adding paragraphs, vertical glue, or page breaks while keeping the signature
% anchored exactly 1.0cm from the bottom edge of the final shipped page.

\fancyfoot[C]{%
  \ifnum\value{page}=\getpagerefnumber{LastPage}\relax
    \begin{tikzpicture}[remember picture, overlay]
      \node[anchor=south] at ([yshift=1.0cm]current page.south) {
        \footnotesize
        \begin{tabular}{c}
          \textbf{TECHNOLOGY IS ELEGANT}\\
          Emir A.\\
          \institution
        \end{tabular}
      };
    \end{tikzpicture}%
  \fi
}
